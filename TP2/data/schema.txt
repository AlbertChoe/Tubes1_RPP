====================================================
RELATIONSHIPS
====================================================

(:BMWModel)-[:HAS_SERIES]->(:Series)  // Maps a model to the official BMW Series line.
(:BMWModel)-[:HAS_BODY_TYPE]->(:BodyType) // Maps a model to its physical body style.
(:BMWModel)-[:HAS_POWERTRAIN]->(:Powertrain) // Maps a model to its power source.
(:BMWModel)-[:HAS_DRIVETRAIN]->(:Drivetrain) // Maps a model to its direction of power delivery.
(:BMWModel)-[:HAS_FEATURE]->(:Feature) // Maps a model to its features.
(:BMWModel)-[:HAS_PERFORMANCE]->(:Performance) // Maps a model to its performance tier.

====================================================
COMMON QUERY PATTERNS
====================================================

-- Find models by body type
MATCH (m:BMWModel)-[:HAS_BODY_TYPE]->(b:BodyType {name:$body})
RETURN m

-- Find electric models
MATCH (m:BMWModel:ElectricCar)
RETURN m

-- Find M Series performance cars
MATCH (m:BMWModel)-[:HAS_PERFORMANCE]->(:Performance {name:"M Series"})
RETURN m

-- Find SUVs with xDrive capability
MATCH (m:BMWModel:OffroadCapableSUV)
RETURN m

-- Find all models in a given series
MATCH (m:BMWModel)-[:HAS_SERIES]->(s:Series {name:$series})
RETURN m

-- Full profile for a model
MATCH (m:BMWModel {name:$model})
OPTIONAL MATCH (m)-[:HAS_SERIES]->(s)
OPTIONAL MATCH (m)-[:HAS_BODY_TYPE]->(b)
OPTIONAL MATCH (m)-[:HAS_POWERTRAIN]->(p)
OPTIONAL MATCH (m)-[:HAS_DRIVETRAIN]->(d)
OPTIONAL MATCH (m)-[:HAS_PERFORMANCE]->(perf)
OPTIONAL MATCH (m)-[:HAS_FEATURE]->(f)
RETURN m, s, b, p, d, perf, collect(f) AS features
